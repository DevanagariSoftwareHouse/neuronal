\chapter{Natural numbers, counting, and induction}\label{ch:naturals}
\begin{readergoal}See how counting can be built from basic rules, rather than accepted as an unexplained operation.\end{readergoal}
\section{Counting precedes measurement}
When we say there were seven action potentials, we count discrete events. Counting does not require time units, voltage calibration, or a model of firing probability. The natural numbers are the first formal setting for such statements. We adopt $\mathbb N_0=\{0,1,2,\ldots\}$, reserving $\mathbb N_+=\{1,2,\ldots\}$. Authors differ about including zero, so every technical treatment should make its convention explicit.

\section{Successors and the Peano viewpoint}
Imagine a set $N$ with a distinguished element $0$ and successor operation $S:N\to N$. The basic idea is that zero is not the successor of anything; distinct inputs have distinct successors; and any subset containing zero and closed under the successor operation contains all of $N$. The final clause is the induction principle. These structural properties establish a unique natural-number system up to isomorphism, under the usual axiomatic background \cite{MendelsonNumbers}.

Define addition recursively by $n+0=n$ and $n+S(m)=S(n+m)$. Define multiplication through $n\cdot0=0$ and $n\cdot S(m)=n\cdot m+n$. These are \emph{definitions}, while commutativity, associativity, and distributivity are derived properties. We ordinarily write $n+1$ rather than $S(n)$ after connecting the symbols.

\section{Induction as a way to prove infinite families of claims}
Suppose we wish to prove $\sum_{k=1}^n k=n(n+1)/2$ for all positive integers $n$. The base case $n=1$ holds. Assume it is true for $n$. Then
\[\sum_{k=1}^{n+1}k=\frac{n(n+1)}2+(n+1)=\frac{(n+1)(n+2)}2.\]
Therefore the claim holds for $n+1$; induction establishes every case. Induction is a proof principle, not an empirical inference from observing a finite number of examples.

\section{Finite sets and combinatorics}
Two finite sets have the same cardinality if there exists a bijection between them. The number of length-$n$ binary vectors is $2^n$; the number of ways to choose $k$ distinct objects from $n$ without order is $\binom nk=n!/[k!(n-k)!]$. If 20 channels are inspected in all unordered pairs, the number of pairs is $\binom{20}{2}=190$. This count matters when reporting a family of connectivity tests.

\begin{workedexample}[Discrete time windows]
A recording sampled at $f_s=1000$ samples/s for 10 seconds has $N=f_sT=10000$ samples if we choose the index set $\{0,\ldots,9999\}$. The sample at index $n$ corresponds to $t_n=n/f_s$, so the last recorded time is $9.999$ s, not 10 s. Distinguish the \emph{duration of acquisition} from the \emph{timestamp of its final sample}.
\end{workedexample}

\section{Proving the arithmetic laws from the definitions}
The recursive definitions of addition and multiplication say nothing explicit about commutativity or associativity. These must be proved by induction, and the proofs show how a small set of rules generates all of school arithmetic.

\begin{proposition}[Associativity of addition]
For all $k,m,n\in\mathbb N_0$, $(k+m)+n=k+(m+n)$.
\end{proposition}
\begin{proof}
Fix $k$ and $m$, and induct on $n$. For $n=0$, both sides equal $k+m$ by the rule $x+0=x$. Suppose the identity holds for $n$. Then
\[
(k+m)+S(n)=S\bigl((k+m)+n\bigr)=S\bigl(k+(m+n)\bigr)=k+S(m+n)=k+\bigl(m+S(n)\bigr),
\]
where the first and third equalities use the recursive rule $x+S(y)=S(x+y)$, the second uses the induction hypothesis, and the last uses the recursive rule once more.
\end{proof}

\begin{proposition}[Commutativity of addition]
For all $m,n\in\mathbb N_0$, $m+n=n+m$.
\end{proposition}
\begin{proof}
We first prove two lemmas by induction on $n$: (a) $0+n=n$, and (b) $S(m)+n=S(m+n)$. For (a), $0+0=0$, and $0+S(n)=S(0+n)=S(n)$. For (b), $S(m)+0=S(m)=S(m+0)$, and $S(m)+S(n)=S(S(m)+n)=S(S(m+n))=S(m+S(n))$. Now induct on $n$ in the main claim. The case $n=0$ is $m+0=m=0+m$ by (a). If $m+n=n+m$, then $m+S(n)=S(m+n)=S(n+m)=S(n)+m$ by (b).
\end{proof}
Analogous inductions establish associativity and commutativity of multiplication and the distributive law $k(m+n)=km+kn$. The order relation is then \emph{defined} by $m\le n$ iff $n=m+p$ for some $p\in\mathbb N_0$, and one proves that it is a total order compatible with addition and multiplication.

\section{Recursion as a way to define sequences}
The \emph{recursion theorem} states that, given a set $X$, an element $x_0\in X$, and a function $g:X\to X$, there is exactly one sequence $x:\mathbb N_0\to X$ with $x(0)=x_0$ and $x(S(n))=g(x(n))$. It is the justification for every iterative definition: factorials, powers, and numerical algorithms such as Newton's method. Uniqueness follows from induction on $n$; existence requires a construction (the intersection of all relations with the right closure properties) which we omit.

\begin{workedexample}[A recursive filter is a recursive definition]
The exponential smoother $y_0=x_0$, $y_{n}=\alpha y_{n-1}+(1-\alpha)x_n$ for $0<\alpha<1$ is defined by recursion. Unrolling by induction gives
\[
y_n=\alpha^nx_0+(1-\alpha)\sum_{k=1}^{n}\alpha^{n-k}x_k,
\]
which can be verified by checking $n=0$ and the step $n\to n+1$. The recursion is computationally cheap; the closed form shows the geometric weighting of past samples. This is the simplest infinite-impulse-response filter, revisited in Chapter~\ref{ch:fourier}.
\end{workedexample}

\section{Divisibility, remainders, and modular indices}
\begin{theorem}[Division with remainder]
For $n\in\mathbb N_0$ and $d\in\mathbb N_+$ there are unique $q,r\in\mathbb N_0$ with $n=qd+r$ and $0\le r<d$.
\end{theorem}
\begin{proof}
Existence: the set $\{n-qd:q\in\mathbb N_0,\ qd\le n\}$ is a nonempty subset of $\mathbb N_0$ (it contains $n$), so by well ordering it has a least element $r=n-qd$. If $r\ge d$, then $n-(q+1)d=r-d\ge0$ is a smaller element, a contradiction. Uniqueness: if $qd+r=q'd+r'$ with $0\le r,r'<d$, then $|r-r'|<d$ is a multiple of $d$, hence $0$, and then $q=q'$.
\end{proof}
Remainders appear constantly in indexing. Converting a linear sample index $n$ into (trial, within-trial sample) for trials of length $L$ is $n=qL+r$; circular buffers use $r=n\bmod L$; and the discrete Fourier transform treats its indices modulo $N$. Off-by-one errors are often confusions between the ranges $\{0,\ldots,L-1\}$ and $\{1,\ldots,L\}$.

\section{Counting principles}
Three counting rules recur:
\begin{enumerate}[leftmargin=2em]
\item \emph{Sum rule}: if a finite set is a disjoint union $A\cup B$, then $|A\cup B|=|A|+|B|$.
\item \emph{Product rule}: $|A\times B|=|A|\,|B|$.
\item \emph{Ordered selections}: the number of injective functions from a $k$-set to an $n$-set is $n(n-1)\cdots(n-k+1)=n!/(n-k)!$.
\end{enumerate}
Each unordered $k$-subset corresponds to exactly $k!$ ordered selections, which gives $\binom nk=n!/[k!(n-k)!]$. Pascal's rule $\binom nk=\binom{n-1}{k-1}+\binom{n-1}{k}$ follows by splitting subsets according to whether they contain a fixed element, and induction with Pascal's rule proves the binomial theorem
\[
(a+b)^n=\sum_{k=0}^n\binom nk a^kb^{n-k}.
\]
With $a=p$ and $b=1-p$, the terms are the binomial probabilities of observing $k$ spikes in $n$ independent bins, and the identity says they sum to one.
